# 🔍 Redirecting...

<p id="msg">Searching for page...</p>

<script>
  (function () {
    var pathname = window.location.pathname;
    var hash = window.location.hash || '';
    var search = window.location.search || '';

    // Convert lowercase path
    var lowerPath = pathname.toLowerCase();

    // If case differed (e.g. /EVENTS.html -> /events.html), redirect to lowercase
    if (pathname !== lowerPath) {
      window.location.replace(lowerPath + search + hash);
      return;
    }

    // Known docs pages
    var knownPages = [
      "index.html", "installation.html", "configuration.html", "features.html",
      "services.html", "events.html", "automations.html", "buttons.html",
      "quoting.html", "missed_messages.html", "autoresponder.html", "moderation.html",
      "webhooks.html", "rocketchat.html", "telegram.html", "tips.html",
      "troubleshooting.html", "limitations.html", "i18n.html", "support.html",
      "beta.html", "api.html", "architecture.html", "development.html"
    ];

    var segments = pathname.split('/');
    var lastSegment = segments[segments.length - 1].toLowerCase();

    if (lastSegment) {
      if (!lastSegment.endsWith('.html')) {
        var withHtml = lastSegment + '.html';
        if (knownPages.indexOf(withHtml) !== -1) {
          segments[segments.length - 1] = withHtml;
          window.location.replace(segments.join('/') + search + hash);
          return;
        }
      }
    }

    document.getElementById('msg').innerHTML = 'The requested page could not be found. <a href="/ha-whatsapp/">Go back to home</a>';
  })();
</script>
