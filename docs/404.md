<p id="msg">Searching for requested page...</p>

<script>
  (function () {
    var rawPath = window.location.pathname;
    var hash = window.location.hash || '';
    var search = window.location.search || '';

    // Extract repo prefix if present (e.g. /ha-whatsapp/)
    var base = '/ha-whatsapp/';
    var rel = rawPath;
    if (rel.indexOf(base) === 0) {
      rel = rel.substring(base.length);
    } else if (rel.indexOf('/') === 0) {
      rel = rel.substring(1);
    }

    var cleanName = rel.toLowerCase().replace(/\/$/, '');
    if (cleanName.endsWith('.html')) {
      cleanName = cleanName.substring(0, cleanName.length - 5);
    }

    // Direct mapping for doc names
    var validPages = [
      "index", "installation", "configuration", "features",
      "services", "events", "automations", "buttons",
      "quoting", "missed_messages", "autoresponder", "moderation",
      "webhooks", "rocketchat", "telegram", "tips",
      "troubleshooting", "limitations", "i18n", "support",
      "beta", "api", "architecture", "development"
    ];

    if (validPages.indexOf(cleanName) !== -1) {
      var target = base + cleanName + '.html' + search + hash;
      window.location.replace(target);
      return;
    }

    document.getElementById('msg').innerHTML = 'The requested page could not be found. <a href="/ha-whatsapp/">Go back to home</a>';
  })();
</script>
